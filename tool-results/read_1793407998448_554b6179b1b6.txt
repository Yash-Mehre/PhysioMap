    78→    78→const VALIDATABLE_FIELDS: readonly ValidatableField[] = [
    79→    79→  'age',
    80→    80→  'sex',

    90→    90→interface HealthFieldValues {

   100→   100→/** Map a zod failure to "first message per field", keyed by field name. */
   101→   101→function validateFields(values: HealthFieldValues): FieldErrors {
   102→   102→  const parsed = healthProfileSchema.safeParse(values)
   103→   103→  if (parsed.success) return {}
   104→   104→  const errors: FieldErrors = {}
   105→   105→  for (const issue of parsed.error.issues) {
   106→   106→    const key = issue.path[0]
   107→   107→    if (typeof key !== 'string') continue
   108→   108→    const field = VALIDATABLE_FIELDS.find((candidate) => candidate === key)
   109→   109→    if (field !== undefined && errors[field] === undefined) {
   110→   110→      errors[field] = issue.message
   111→   111→    }
   112→   112→  }
   113→   113→  return errors
   114→   114→}
   115→   115→
   116→   116→/** Parse a numeric input string; blank or non-finite becomes undefined. */
   117→   117→function parseNumberInput(value: string): number | undefined {
   118→   118→  const trimmed = value.trim()
   119→   119→  if (trimmed === '') return undefined
   120→   120→  const parsed = Number(trimmed)
   121→   121→  return Number.isFinite(parsed) ? parsed : undefined
   122→   122→}
   123→   123→
   124→   124→/** Feet + inches input strings → centimetres (1 decimal); undefined when both blank. */
   125→   125→function imperialToCm(feetStr: string, inchesStr: string): number | undefined {
   126→   126→  if (feetStr.trim() === '' && inchesStr.trim() === '') return undefined
   127→   127→  const feet = feetStr.trim() === '' ? 0 : Number(feetStr)
   128→   128→  const inches = inchesStr.trim() === '' ? 0 : Number(inchesStr)
   129→   129→  if (!Number.isFinite(feet) || !Number.isFinite(inches)) return undefined
   130→   130→  return Math.round(ftInToCm(feet, inches) * 10) / 10
   131→   131→}
   132→   132→
   133→   133→/** Pounds input string → kilograms (1 decimal); undefined when blank. */
   134→   134→function poundsToKg(lbStr: string): number | undefined {
   135→   135→  const lb = parseNumberInput(lbStr)
   136→   136→  return lb === undefined ? undefined : Math.round(lbToKg(lb) * 10) / 10
   137→   137→}
   138→   138→
   139→   139→/** Inline validation message for a field (first zod issue only). */
   140→   140→function FieldError({ id, message }: { id: string; message: string | undefined }) {
   141→   141→  if (message === undefined) return null
   142→   142→  return (
   143→   143→    <p id={id} aria-live="polite" className="text-xs text-mode-pain">
   144→   144→      {message}
   145→   145→    </p>
   146→   146→  )
   147→   147→}

   314→   314→        {/* ---------------------------------------------------------- Body stats */}
   315→   315→        <section className="space-y-3" aria-label="Body stats">

   317→   317→          <div className="grid grid-cols-2 gap-3">

   321→   321→                id={`${formId}-name`}
   322→   322→                type="text"
   323→   323→                maxLength={PROFILE_NAME_MAX_LENGTH}
   324→   324→                value={nameStr}
