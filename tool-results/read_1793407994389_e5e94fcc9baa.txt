    68→/** Fields of `healthProfileSchema` that the form validates inline. */
    69→type ValidatableField =
    70→  | 'age'
    71→  | 'sex'
    72→  | 'heightCm'
    73→  | 'weightKg'
    74→  | 'activityLevel'
    75→  | 'goal'

    78→const VALIDATABLE_FIELDS: readonly ValidatableField[] = [
    79→  'age',
    80→  'sex',
    81→  'heightCm',
    82→  'weightKg',
    83→  'activityLevel',
    84→  'goal',
    85→  'paceKgPerWeek',
    86→]
    87→
    88→type FieldErrors = Partial<Record<ValidatableField, string>>
    89→
    90→interface HealthFieldValues {
    91→  age?: number
    92→  sex?: Sex
    93→  heightCm?: number
    94→  weightKg?: number
    95→  activityLevel?: ActivityLevel
    96→  goal?: WeightGoal
    97→  paceKgPerWeek?: PaceKgPerWeek
    98→}
    99→
   100→/** Map a zod failure to "first message per field", keyed by field name. */
   101→function validateFields(values: HealthFieldValues): FieldErrors {
   102→  const parsed = healthProfileSchema.safeParse(values)
   103→  if (parsed.success) return {}
   104→  const errors: FieldErrors = {}
   105→  for (const issue of parsed.error.issues) {
   106→    const key = issue.path[0]
   107→    if (typeof key !== 'string') continue
   108→    const field = VALIDATABLE_FIELDS.find((candidate) => candidate === key)
   109→    if (field !== undefined && errors[field] === undefined) {
   110→      errors[field] = issue.message
   111→    }
   112→  }
   113→  return errors
   114→}
   115→
   116→/** Parse a numeric input string; blank or non-finite becomes undefined. */
   117→function parseNumberInput(value: string): number | undefined {
   118→  const trimmed = value.trim()
   119→  if (trimmed === '') return undefined
   120→  const parsed = Number(trimmed)
   121→  return Number.isFinite(parsed) ? parsed : undefined
   122→}
   123→
   124→/** Feet + inches input strings → centimetres (1 decimal); undefined when both blank. */
   125→function imperialToCm(feetStr: string, inchesStr: string): number | undefined {
   126→  if (feetStr.trim() === '' && inchesStr.trim() === '') return undefined
   127→  const feet = feetStr.trim() === '' ? 0 : Number(feetStr)
   128→  const inches = inchesStr.trim() === '' ? 0 : Number(inchesStr)
   129→  if (!Number.isFinite(feet) || !Number.isFinite(inches)) return undefined
   130→  return Math.round(ftInToCm(feet, inches) * 10) / 10
   131→}
   132→
   133→/** Pounds input string → kilograms (1 decimal); undefined when blank. */
   134→function poundsToKg(lbStr: string): number | undefined {
   135→  const lb = parseNumberInput(lbStr)
   136→  return lb === undefined ? undefined : Math.round(lbToKg(lb) * 10) / 10
   137→}
   138→
   139→/** Inline validation message for a field (first zod issue only). */
   140→function FieldError({ id, message }: { id: string; message: string | undefined }) {
   141→  if (message === undefined) return null
   142→  return (
   143→    <p id={id} aria-live="polite" className="text-xs text-mode-pain">
   144→      {message}
   145→    </p>
   146→  )
   147→}

   161→  const [ageStr, setAgeStr] = useState(initialProfile.age !== undefined ? String(initialProfile.age) : '')

   164→  const [heightCmStr, setHeightCmStr] = useState(

   166→  )
   167→  const [weightKgStr, setWeightKgStr] = useState(

   169→  )

   171→    initialProfile.heightCm !== undefined ? String(cmToFtIn(initialProfile.heightCm).feet) : '',
   172→  )
   173→  const [heightInStr, setHeightInStr] = useState(() =>
   174→    initialProfile.heightCm !== undefined ? String(cmToFtIn(initialProfile.heightCm).inches) : '',
   175→  )
   176→  const [weightLbStr, setWeightLbStr] = useState(() =>
   177→    initialProfile.weightKg !== undefined ? String(Math.round(kgToLb(initialProfile.weightKg))) : '',
   178→  )

   180→  const [goal, setGoal] = useState<WeightGoal | undefined>(initialProfile.goal)
   181→  const [pace, setPace] = useState<PaceKgPerWeek>(initialProfile.paceKgPerWeek ?? 0.5)
   182→  const [dietPreference, setDietPreference] = useState<DietPreference>(initialProfile.dietPreference)
   183→  const [excludedTags, setExcludedTags] = useState<DietTag[]>(initialProfile.excludedTags)
   184→  const [trainingLevel, setTrainingLevel] = useState<Level>(initialProfile.trainingLevel)
   185→  const [trainingGoal, setTrainingGoal] = useState<TrainingGoal>(initialProfile.trainingGoal)
   186→  const [equipment, setEquipment] = useState<Equipment[]>(initialProfile.equipment)
   187→  const [medicalConditions, setMedicalConditions] = useState<MedicalConditionId[]>(

   189→  )
   190→  const [otherConditions, setOtherConditions] = useState<string[]>(initialProfile.otherConditions)
   191→
   192→  // Always-metric values (imperial inputs convert through the helpers above).
   193→  const age = parseNumberInput(ageStr)
   194→  const heightCm = unitSystem === 'metric' ? parseNumberInput(heightCmStr) : imperialToCm(heightFtStr, heightInStr)
   195→  const weightKg = unitSystem === 'metric' ? parseNumberInput(weightKgStr) : poundsToKg(weightLbStr)
   196→
   197→  const errors = validateFields({ age, sex, heightCm, weightKg, activityLevel, goal, paceKgPerWeek: pace })
   198→  const trimmedName = nameStr.trim()
   199→  const nameError =
   200→    trimmedName.length > PROFILE_NAME_MAX_LENGTH
   201→      ? `Name must be ${PROFILE_NAME_MAX_LENGTH} characters or fewer.`
   202→      : undefined
   203→  const isValid = Object.keys(errors).length === 0 && nameError === undefined
   204→
   205→  const draft: UserProfile = {
   206→    ...initialProfile,
   207→    name: trimmedName === '' ? undefined : trimmedName,
   208→    age,
   209→    sex,
   210→    heightCm,
   211→    weightKg,
   212→    activityLevel,
   213→    goal,
   214→    paceKgPerWeek: pace,
   215→    unitSystem,
   216→    dietPreference,
   217→    excludedTags,
   218→    trainingLevel,
   219→    trainingGoal,
   220→    equipment,
   221→    medicalConditions,
   222→    otherConditions,
   223→  }
   224→
   225→  const handleSexChange = (value: string): void => {
   226→    const next = SEX_OPTIONS.find((option) => option === value)
   227→    if (next !== undefined) setSex(next)
   228→  }
   229→
   230→  const handleActivityChange = (value: string): void => {
   231→    const next = ACTIVITY_LEVEL_ORDER.find((option) => option === value)
   232→    if (next !== undefined) setActivityLevel(next)
   233→  }
   234→
   235→  const handleGoalChange = (value: string): void => {
   236→    const next = GOAL_OPTIONS.find((option) => option === value)
   237→    if (next !== undefined) setGoal(next)
   238→  }
   239→
   240→  const handlePaceChange = (value: string): void => {
   241→    const next = PACE_OPTIONS.find((option) => String(option) === value)
   242→    if (next !== undefined) setPace(next)
   243→  }
   244→
   245→  const handleDietChange = (value: string): void => {
   246→    const next = DIET_OPTIONS.find((option) => option === value)
   247→    if (next !== undefined) setDietPreference(next)
   248→  }
   249→
   250→  const handleTrainingLevelChange = (value: string): void => {
   251→    const next = LEVEL_OPTIONS.find((option) => option === value)
   252→    if (next !== undefined) setTrainingLevel(next)
   253→  }
   254→
   255→  const handleTrainingGoalChange = (value: string): void => {
   256→    const next = TRAINING_GOAL_OPTIONS.find((option) => option === value)
   257→    if (next !== undefined) setTrainingGoal(next)
   258→  }
   259→
   260→  /** Toggle units, converting the visible buffers in both directions. */

   262→    if (next === unitSystem) return
   263→    if (next === 'imperial') {

   266→        const split = cmToFtIn(cm)
   267→        setHeightFtStr(String(split.feet))
   268→        setHeightInStr(String(split.inches))
   269→      } else {
   270→        setHeightFtStr('')
   271→        setHeightInStr('')
   272→      }
   273→      const kg = parseNumberInput(weightKgStr)
   274→      setWeightLbStr(kg !== undefined ? String(Math.round(kgToLb(kg))) : '')
   275→    } else {
   276→      const cm = imperialToCm(heightFtStr, heightInStr)
   277→      setHeightCmStr(cm !== undefined ? String(cm) : '')
   278→      const kg = poundsToKg(weightLbStr)
   279→      setWeightKgStr(kg !== undefined ? String(kg) : '')
   280→    }
   281→    setUnitSystem(next)
   282→  }
   283→
   284→  const toggleTag = (tag: DietTag): void => {
   285→    setExcludedTags((current) =>
   286→      current.includes(tag) ? current.filter((item) => item !== tag) : [...current, tag],
   287→    )
   288→  }
   289→
   290→  const toggleEquipment = (item: Equipment): void => {
   291→    setEquipment((current) => {
   292→      const next = current.includes(item) ? current.filter((value) => value !== item) : [...current, item]
   293→      return next.length > 0 ? next : ['bodyweight']
   294→    })
   295→  }

   297→  const heightHint =
   298→    unitSystem === 'imperial' && heightCm !== undefined ? `≈ ${heightCm} cm` : undefined

   300→    unitSystem === 'imperial' && weightKg !== undefined ? `≈ ${weightKg} kg` : undefined
   301→  const heightMessage = errors.heightCm ?? heightHint
   302→  const weightMessage = errors.weightKg ?? weightHint
   303→
   304→  return (

   307→      onSubmit={(event) => {

   311→      className="flex h-full min-h-0 flex-col"
   312→    >
   313→      <div className="scroll-slim min-h-0 flex-1 space-y-5 overflow-y-auto px-4 py-4 sm:px-5">

   316→          <h3 className="text-xs font-semibold uppercase tracking-[0.16em] text-copper">Body stats</h3>
   317→          <div className="grid grid-cols-2 gap-3">
   318→            <div className="col-span-2 space-y-1.5">
   319→              <Label htmlFor={`${formId}-name`}>Name (optional)</Label>
   320→              <Input
   321→                id={`${formId}-name`}
   322→                type="text"
   323→                maxLength={PROFILE_NAME_MAX_LENGTH}
   324→                value={nameStr}
   325→                onChange={(event) => setNameStr(event.target.value)}
   326→                placeholder="Shown in your profile header"
   327→                autoComplete="name"
   328→                aria-invalid={nameError !== undefined}
   329→                aria-describedby={nameError !== undefined ? `${formId}-name-error` : undefined}
   330→              />
   331→              <FieldError id={`${formId}-name-error`} message={nameError} />
   332→            </div>
   333→            <div className="space-y-1.5">
   334→              <Label htmlFor={`${formId}-age`}>Age</Label>
   335→              <Input
   336→                id={`${formId}-age`}
   337→                type="number"
   338→                inputMode="numeric"
   339→                min={10}
   340→                max={100}
   341→                value={ageStr}
   342→                onChange={(event) => setAgeStr(event.target.value)}
   343→                aria-invalid={errors.age !== undefined}
   344→                aria-describedby={errors.age !== undefined ? `${formId}-age-error` : undefined}
   345→              />
   346→              <FieldError id={`${formId}-age-error`} message={errors.age} />
   347→            </div>
   348→            <div className="space-y-1.5">
   349→              <Label htmlFor={`${formId}-sex`}>Sex</Label>
   350→              <Select value={sex} onValueChange={handleSexChange}>
   351→                <SelectTrigger
   352→                  id={`${formId}-sex`}
   353→                  className="w-full"
   354→                  aria-invalid={errors.sex !== undefined}
   355→                  aria-describedby={errors.sex !== undefined ? `${formId}-sex-error` : undefined}
   356→                >
   357→                  <SelectValue placeholder="Select" />
   358→                </SelectTrigger>
   359→                <SelectContent>
   360→                  {SEX_OPTIONS.map((option) => (
   361→                    <SelectItem key={option} value={option}>
   362→                      {SEX_LABELS[option]}
   363→                    </SelectItem>
   364→                  ))}
   365→                </SelectContent>
   366→              </Select>
   367→              <FieldError id={`${formId}-sex-error`} message={errors.sex} />
   368→            </div>
   369→            <div className="col-span-2 space-y-1.5">
   370→              <span id={`${formId}-units-label`} className="block text-sm font-medium leading-none">
   371→                Units
   372→              </span>
   373→              <div
   374→                role="group"
   375→                aria-labelledby={`${formId}-units-label`}
   376→                className="grid w-full grid-cols-2 gap-1 rounded-lg border border-white/8 bg-surface-raised p-1"

   378→                {UNIT_SYSTEM_OPTIONS.map((option) => (
   379→                  <button
   380→                    key={option}
   381→                    type="button"
   382→                    aria-pressed={unitSystem === option}
   383→                    onClick={() => handleUnitChange(option)}
   384→                    className={cn(
   385→                      'min-h-9 rounded-md px-3 text-xs font-semibold transition-colors duration-150 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring/60',
   386→                      unitSystem === option

   388→                        : 'text-muted-foreground hover:text-foreground',

   390→                  >

   478→                    step="any"
   479→                    value={weightLbStr}
   480→                    onChange={(event) => setWeightLbStr(event.target.value)}
   481→                    aria-invalid={errors.weightKg !== undefined}
   482→                    aria-describedby={weightMessage !== undefined ? `${formId}-weight-error` : undefined}
   483→                  />
   484→                  {weightMessage !== undefined && (
   485→                    <p
   486→                      id={`${formId}-weight-error`}
   487→                      aria-live="polite"
   488→                      className={cn(
   489→                        'text-xs',
   490→                        errors.weightKg !== undefined ? 'text-mode-pain' : 'text-muted-foreground',
   491→                      )}
   492→                    >
   493→                      {weightMessage}
   494→                    </p>
   495→                  )}
   496→                </div>
   497→              </>

   499→          </div>

   501→
   502→        {/* --------------------------------------------------- Activity & goal */}
